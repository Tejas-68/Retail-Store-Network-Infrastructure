# ==============================================================================
# CISCO IOS SWITCH CONFIGURATIONS: RETAIL STORE INFRASTRUCTURE
# ==============================================================================
# Architecture: Hierarchical 2-Tier Model (Distribution / Access)
# Devices Included:
#   1. SW-CORE   - Cisco Catalyst 3560/3650 Distribution Switch (MDF Rack 1)
#   2. SW-POS    - Cisco Catalyst 2960 Access Switch (IDF 1 - Cashier Bank)
#   3. SW-STAFF  - Cisco Catalyst 2960 Access Switch (IDF 2 - Store Admin)
#   4. SW-SERVER - Cisco Catalyst 3750/2960 Access Switch (MDF Rack 2)
# ==============================================================================

################################################################################
### SECTION 1: CORE / DISTRIBUTION SWITCH (SW-CORE)                          ###
################################################################################
hostname SW-CORE
enable secret 5 $1$mERr$hx5rVt7rPNoS4wqbXKX7m0
service password-encryption
no ip domain-lookup
ip domain-name retail.internal
ip ssh version 2
username admin privilege 15 secret 5 $1$mERr$hx5rVt7rPNoS4wqbXKX7m0

! STP Rapid-PVST Root Configuration
spanning-tree mode rapid-pvst
spanning-tree extend system-id
spanning-tree vlan 10,20,30,40,99 root primary

! VLAN Creation
vlan 10
 name POS-TERMINALS
vlan 20
 name STAFF-WORKSTATIONS
vlan 30
 name SERVER-FARM
vlan 40
 name NETWORK-PRINTERS
vlan 99
 name MANAGEMENT-SVI
vlan 666
 name NATIVE-BLACKHOLE

! Router Uplink (802.1Q Trunk)
interface GigabitEthernet0/1
 description Trunk to R1-EDGE [Gi0/0/0]
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 666
 switchport trunk allowed vlan 10,20,30,40,99
 switchport nonegotiate
 duplex full
 speed 1000
 no shutdown

! Downlink to SW-POS
interface GigabitEthernet0/2
 description Trunk to SW-POS [Gi0/1]
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 666
 switchport trunk allowed vlan 10,99
 switchport nonegotiate
 duplex full
 speed 1000
 no shutdown

! Downlink to SW-STAFF
interface GigabitEthernet0/3
 description Trunk to SW-STAFF [Gi0/1]
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 666
 switchport trunk allowed vlan 20,40,99
 switchport nonegotiate
 duplex full
 speed 1000
 no shutdown

! Downlink to SW-SERVER
interface GigabitEthernet0/4
 description Trunk to SW-SERVER [Gi0/1]
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 666
 switchport trunk allowed vlan 30,99
 switchport nonegotiate
 duplex full
 speed 1000
 no shutdown

! Shutdown Unused Ports
interface range GigabitEthernet0/5 - 24
 description UNUSED PORTS - SECURITY DISABLED
 switchport mode access
 switchport access vlan 666
 shutdown

! Management SVI
interface Vlan99
 description SW-CORE Management Interface
 ip address 192.168.99.2 255.255.255.0
 no shutdown
ip default-gateway 192.168.99.1

banner motd ^C
*******************************************************************************
* WARNING: AUTHORIZED ACCESS ONLY - RETAIL CORE DISTRIBUTION SWITCH (SW-CORE) *
*******************************************************************************
^C
line con 0
 exec-timeout 5 0
 logging synchronous
 login local
line vty 0 4
 exec-timeout 10 0
 logging synchronous
 login local
 transport input ssh
!
